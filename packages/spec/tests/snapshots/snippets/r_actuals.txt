library(httr2)

resp <- request('https://api.mapi.example/v1/endpoints/sales-forecast/score') |>
  req_auth_bearer_token(Sys.getenv('MAPI_TOKEN')) |>
  req_headers(`X-Mapi-Project` = 'acme/growth', Prefer = 'wait=60') |>
  req_body_raw('{
  "mode": "forecast",
  "horizon": {
    "dim": "t",
    "periods": 8
  },
  "data": null,
  "scenario": null,
  "output": {
    "summary": [
      "median",
      "q5",
      "q95"
    ]
  },
  "future_covariates": "carry_last"
}', type = "application/json") |>
  req_perform()
result <- resp_body_json(resp)
str(result$run$state)
# the numbers: one table per output (coordinates, then mean and HDI)
str(result$summary$tables)

# Forecast with your latest actuals and the dates to forecast: in 'y' put the actuals you
# have for the periods after the fitted history and leave null on the periods to forecast
# (driver values left empty there are carried forward). The model updates its state with the
# actuals - it is not refitted - and summary.conditioning in the answer says what the
# forecast was conditioned on.
