    A contract kit runs your extension the way the platform will, and checks the rules that the platform depends on. It tells you about a problem before anybody else runs your code. Every scaffold has one kit test for each kind, marked mapi_contract; mapi plugin test runs them, and mapi plugin test --contract-only runs only them.

    Kind Kit Import from Returns
    Model check_model(plugin, fixture, *, require_about=False, raise_on_failure=True, checks=None) mapi_sdk.testing ContractReport
    Step check_step(step, ds, settings, *, schema=None, require_about=False, raise_on_failure=True) mapi_sdk.testing ContractReport
    Plot recipe check_recipe(recipe, *, require_about=True, raise_on_failure=True) mapi_plots.testing RecipeReport
    Workflow node check_node(export, *, inputs, params=None, seed=0, raise_on_failure=True) mapi_workflows.kit ContractReport
    Workflow template check_template(t, *, nodes=None, warnings=False) mapi_workflows.kit list[Issue]

    With raise_on_failure=False, a kit returns its report: report.ok and report.summary(), one line for each check. Pass require_about=True for the model and step kits, as the scaffold does: the catalog shows the About card, and the loader marks an incomplete card with about_incomplete.

    mapi_workflows.testing re-exports check_node and check_template, but it needs the testing extra and imports the statsmodels registry. Import them from mapi_workflows.kit.

    Models¶

    """Contract-kit tests for acme.group_mean (run them with ``mapi plugin test``).

The kit fits the model on a small daily panel (resampled to weeks by a toy step), then
checks bundle round trips (no pickle), shuffled/subset/new-level scoring, typed errors for
every dim policy, context-window equivalence, evaluate == score path, output units,
aggregation legality, seed determinism and the About card. Every check must pass before you
share the plugin.
"""

from __future__ import annotations

import pytest
from mapi_sdk.testing import ModelFixture, check_model, toy

from acme.model import MODELS, GroupMeanModel

SPEC = """
apiVersion: mapi/v1
kind: ModelSpec
metadata: {name: contract-kit, project: test}
data:
  dataset: dataset://test/panel@latest
  bind:
    dims: {geo: market, time: date}
    vars: {y: revenue_local}
  split: {kind: time_holdout, dim: time, last: 4, after_step: weekly}
pipeline:
  - {id: weekly, use: toy.resample@^1, with: {dim: time, freq: W-SUN}}
model:
  use: acme.group_mean@^0.1
  settings: {shrinkage: 0.1}
  fit: {mode: evaluate_then_refit, seed: 7}
evaluate: {metrics: [rmse, mape]}
"""


def test_the_model_is_exported() -> None:
    assert GroupMeanModel in MODELS
    assert GroupMeanModel.catalog_entry().ref == "acme.group_mean@0.1.0"


@pytest.mark.mapi_contract
def test_contract_kit() -> None:
    data, _ = toy.panel_dataset(days=7 * 24)
    fixture = ModelFixture(spec=SPEC, data=data[["revenue_local"]], registry=toy.toy_registry())
    report = check_model(GroupMeanModel, fixture, raise_on_failure=False, require_about=True)
    assert report.ok, report.summary()

    ModelFixture holds a small spec and data; mapi_sdk.testing.toy has panels and a registry with the toy.* steps that a fixture spec may use. The checks: fit and score on the fixture, bundle round trips (no pickle), scoring in another order, on a subset and with new levels, typed errors for each dim policy, context-window equivalence, the evaluate path, output units, aggregation legality, seed determinism and the About card. checks= runs a subset.

    Steps¶

    """Contract-kit tests for acme.winsorize (run them with ``mapi plugin test``).

The kit checks that fitting is deterministic, ``transform`` is pure and never learns from
its input, the fitted state round-trips through a bundle, the declared schema effect
matches the output, an absent target is tolerated and the About card is complete.
"""

from __future__ import annotations

import numpy as np
import pandas as pd
import pytest
import xarray as xr
from mapi_sdk import StepContext, infer_schema
from mapi_sdk.testing import check_step

from acme.steps import STEPS, WinsorizeStep


def store_sales() -> xr.Dataset:
    """Weekly sales of four stores with one extreme week."""
    rng = np.random.default_rng(3)
    time = pd.date_range("2025-01-05", periods=30, freq="W-SUN")
    sales = 200.0 + rng.normal(0, 10, (4, 30))
    sales[1, 7] = 10_000.0
    return xr.Dataset(
        {"y": (("geo", "time"), sales.copy()), "sales": (("geo", "time"), sales)},
        coords={"geo": ["north", "south", "east", "west"], "time": time},
    )


def test_the_step_is_exported() -> None:
    assert WinsorizeStep in STEPS
    assert WinsorizeStep.catalog_entry().id == "acme.winsorize"


def test_the_extreme_week_is_capped() -> None:
    data = store_sales()
    step = WinsorizeStep()
    s = step.parse_settings({"vars": ["sales"], "upper": 0.95})
    ctx = StepContext(step_id="w", phase="fit", schema=infer_schema(data), side_tables={})
    out = step.transform(data, s, step.fit(data, s, ctx), ctx)
    assert float(out["sales"].max()) < 1_000.0


@pytest.mark.mapi_contract
def test_contract_kit() -> None:
    data = store_sales()
    report = check_step(
        WinsorizeStep,
        data,
        {"vars": ["sales"]},
        schema=infer_schema(data, targets=["y"]),
        raise_on_failure=False,
        require_about=True,
    )
    assert report.ok, report.summary()

    The checks: a deterministic fit, a pure transform that never learns from its input, a state that survives a bundle round trip, an output that matches the declared effect, an absent target, and the About card.

    Plot recipes¶

    """Contract-kit tests for acme.weekly_total (run them with ``mapi plugin test``).

The kit checks the catalog entry and namespace, the About card, the settings schema (safe,
small, defaults valid, unknown keys refused) and the catalog example: it renders, its
datasets are inline, the sanitizer accepts it unchanged and it has a caption.
"""

from __future__ import annotations

import pytest
from mapi_plots.testing import check_recipe

from acme.plots import RECIPES, weekly_total


def test_the_recipe_is_exported() -> None:
    assert weekly_total in RECIPES
    assert weekly_total.catalog_entry().id == "acme.weekly_total"


def test_settings_change_the_example() -> None:
    spec = weekly_total.example_spec({"title": "Store totals"})
    assert spec is not None and spec.title == "Store totals"


@pytest.mark.mapi_contract
def test_contract_kit() -> None:
    report = check_recipe(weekly_total, raise_on_failure=False)
    assert report.ok, report.summary()

    The checks, each with its own name in the report: the catalog entry and the namespace, the About card, the settings schema (safe, at most 16 KiB after sanitizing, defaults valid, an unknown key refused), and the catalog example (it renders, its datasets are inline, the sanitizer accepts it unchanged, it has a caption). A colour that is not in mapi_plots.palette is a warning.

    Workflow nodes¶

    """Contract-kit tests for test.acme_zero_share (run them with ``mapi plugin test``).

The kit checks the contract (plugin rules, About card, safe parameter schema), runs the
node on the inputs below, then checks the outputs against their port types, the result
fields, the size of the result sidecar and, for a cacheable node, that two runs agree.
"""

from __future__ import annotations

import numpy as np
import pandas as pd
import pytest
import xarray as xr
from mapi_workflows.kit import check_node

from acme.nodes import NODES, ZERO_SHARE


def weekly_sales(zeros: int = 9) -> xr.Dataset:
    """Sixty weeks of sales, the first ``zeros`` weeks at zero."""
    rng = np.random.default_rng(7)
    time = pd.date_range("2024-01-07", periods=60, freq="W-SUN")
    sales = 100.0 + rng.normal(0, 5, 60)
    sales[:zeros] = 0.0
    return xr.Dataset({"sales": (("time",), sales)}, coords={"time": time})


def test_the_node_is_exported() -> None:
    assert ZERO_SHARE in NODES
    assert ZERO_SHARE.contract.id == "test.acme_zero_share"


@pytest.mark.mapi_contract
def test_contract_kit() -> None:
    report = check_node(
        ZERO_SHARE,
        inputs={"data": weekly_sales()},
        params={"var": "sales", "threshold": 0.1},
        raise_on_failure=False,
    )
    assert report.ok, report.summary()

    The checks: the plugin rules of the contract (id, kind, sandbox, ports, parameter schema, About card), one run on inputs with params, the outputs against their port types, the result fields, the size of the result and, for a cacheable node, that two runs agree.

    Workflow templates¶

    """Contract-kit tests for acme_weekly_check (run them with ``mapi plugin test``).

The kit checks the id (your namespace, not a built-in template id), the About card and the
document, validated against the node contracts (the built-in nodes and this package's node).
"""

from __future__ import annotations

import pytest
from mapi_workflows.kit import check_template
from mapi_workflows.runtime.nodes import all_node_contracts

from acme.nodes import ZERO_SHARE
from acme.templates import TEMPLATES, WEEKLY_CHECK


def test_the_template_is_exported() -> None:
    assert WEEKLY_CHECK in TEMPLATES
    assert WEEKLY_CHECK.id == "acme_weekly_check"


@pytest.mark.mapi_contract
def test_contract_kit() -> None:
    issues = check_template(WEEKLY_CHECK, nodes=all_node_contracts([ZERO_SHARE.contract]))
    assert issues == [], [f"{i.pointer}: {i.message}" for i in issues]

    The checks: the id is in your namespace and is not a built-in template id, the About card, and the document validated against nodes (the built-in contracts when you pass none). An empty list means the template is valid.

    Run them in CI¶

    mapi plugin test --contract-only
mapi --json plugin validate > report.json   # exit 0, 1 or 2; the report as JSON

    mapi plugin validate runs the same checks as the loader of the runner, in a subprocess, on the package's entry points. In v0.19, the same kits will run in the sandbox when a plugin is uploaded.
