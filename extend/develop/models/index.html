    • Use it

    A model plugin is a class with fit and score. The platform does the rest: it binds the dataset's names to your canonical names, splits the data, runs the pipeline, keeps the coordinates, writes the bundle and makes the summaries. Start from the scaffold:

    mapi plugin init acme --kind model                # a CustomModel (numpy, xarray)
mapi plugin init acme --kind model --base pymc    # a PyMCModel (you write build())
mapi plugin init acme --kind model --base sklearn # an SklearnModel (an estimator factory)

    The scaffold¶

    src/acme/model.py (the custom base) is a complete, small model: each group's mean of the target, optionally shrunk toward the global mean.

    """acme.group_mean: a baseline model scaffolded by ``mapi plugin init``.

``GroupMeanModel`` predicts each group's training mean of the target (optionally shrunk
toward the global mean); new groups get the global mean (``pooled``) or zero. It is a
minimal but complete :class:`~mapi_sdk.models.CustomModel`: replace ``fit``/``score`` with
your model, update the About card and keep ``mapi plugin test`` (the contract kit) green.

The platform owns coordinates: binding dataset names to the canonical names below
(``geo``, ``time``, ``y``), reindexing score data to the fit order, flagging new levels
(``geo__is_new``), inverse transforms and reverse binding of the outputs.
"""

from __future__ import annotations

from typing import Any, ClassVar

import numpy as np
import xarray as xr
from mapi_sdk.models import (
    PREDICTIONS,
    CustomModel,
    FitContext,
    FitOutput,
    Fitted,
    ModelSettings,
    ScoreContext,
    new_level_mask,
)
from mapi_spec import Capabilities, Representation
from mapi_spec.about import About
from mapi_spec.spec import FitConfig, ScoreRequest
from pydantic import Field

__all__ = ["MODELS", "GroupMeanModel", "GroupMeanSettings"]

GROUP, TIME, TARGET = "geo", "time", "y"


class GroupMeanSettings(ModelSettings):
    """Validated settings; exported as JSON Schema for the editor and the API."""

    shrinkage: float = Field(
        0.0, ge=0.0, le=1.0, description="Weight of the global mean in each group's level"
    )


class GroupMeanModel(CustomModel[GroupMeanSettings]):
    id = "acme.group_mean"
    version = "0.1.0"
    description = "Per-group mean of the target (a baseline to start from)"
    Settings = GroupMeanSettings
    capabilities = Capabilities(
        inference="point",
        representation=Representation.POINT,
        score_modes=frozenset({"predict", "forecast", "evaluate"}),
    )
    # The catalog card analysts read (Catalog, Help me choose, the model page). Plain words:
    # what it does for them, when to use it and when not to.
    about: ClassVar[About | None] = About(
        title="Group mean baseline",
        summary_plain="Predicts each group with its own average. Use it as a simple baseline "
        "to beat.",
        use_when=["You need a quick baseline before you fit a real model"],
        avoid_when=["You must explain what drives the target"],
        data_needs=["A numeric target per group and period"],
        assumptions=["Each group's level does not change over time"],
        outputs=["One predicted value per group and period"],
        difficulty="beginner",
        typical_runtime="instant",
        example="A store that sold 200 a week on average is predicted to sell 200 next week.",
    )
    # the data the model accepts, in canonical names: one categorical group dim whose new
    # levels are pooled, one time dim that may be extended, and a float target
    schema_template: ClassVar[dict[str, Any]] = {
        "template": {
            "dims": {
                GROUP: {
                    "kind": "categorical",
                    "policy": "new_levels",
                    "new_level_strategy": "pooled",
                },
                TIME: {"kind": "time", "policy": "extend"},
            },
            "vars": {
                TARGET: {
                    "dims": [GROUP, TIME],
                    "dtype": "float64",
                    "role": "target",
                    "missing": "allow",
                }
            },
        }
    }
    outputs: ClassVar[dict[str, Any]] = {"y_hat": {"dims": [GROUP, TIME], "units_of": TARGET}}
    pad_extend = False  # predictions do not depend on the position along time

    def fit(
        self, ds: xr.Dataset, s: GroupMeanSettings, cfg: FitConfig, ctx: FitContext
    ) -> FitOutput:
        y = ds[TARGET]
        global_level = float(y.mean(skipna=True))
        level = (1.0 - s.shrinkage) * y.mean(TIME, skipna=True) + s.shrinkage * global_level
        ctx.progress(1.0, "group means computed")
        params = xr.Dataset({"level": level, "global_level": ((), global_level)})
        return FitOutput(groups={"params": params})

    def score(
        self, fitted: Fitted, ds: xr.Dataset, req: ScoreRequest, ctx: ScoreContext
    ) -> xr.DataTree:
        params = fitted.group("params")
        groups = ds[GROUP].values
        known = params["level"].reindex({GROUP: groups}).values
        is_new = np.asarray(new_level_mask(ds, GROUP).values, dtype=bool) | np.isnan(known)
        fill = 0.0 if ctx.strategy(GROUP) == "zero" else float(params["global_level"])
        level = np.where(is_new, fill, known)
        values = np.repeat(level[:, None], ds.sizes[TIME], axis=1)
        y_hat = xr.DataArray(
            values,
            dims=[GROUP, TIME],
            coords={GROUP: groups, TIME: ds[TIME].values},
        )
        return xr.DataTree.from_dict({PREDICTIONS: xr.Dataset({"y_hat": y_hat})})


MODELS = [GroupMeanModel]
"""The ``mapi.models`` entry point: every model class listed here is registered."""

    What to keep when you replace the example with your model:

    • id is <your namespace>.<name> and version is semver. A spec names it as acme.group_mean@^0.1.
    • Settings is a Pydantic model. Its JSON Schema becomes the settings form in the app, so it must pass the schema-safety rules.
    • about is the card that analysts read in the Catalog and in Help me choose.
    • schema_template declares the data that the model accepts, in canonical names, with a policy for each dim (here: new geo levels are pooled; time may extend).
    • outputs lists each variable that score returns, with its dims.
    • MODELS is what the mapi.models entry point exports.

    The full rules, with PyMC and scikit-learn examples, are in Write a step or model plugin and Port a model in 15 minutes.

    The entry point¶

    pyproject.toml registers the module:

    [project.entry-points."mapi.models"]
acme = "acme.model:MODELS"

    Each kind has its own entry point and module, so a model that fails to import never hides your plots or nodes.

    The contract kit¶

    """Contract-kit tests for acme.group_mean (run them with ``mapi plugin test``).

The kit fits the model on a small daily panel (resampled to weeks by a toy step), then
checks bundle round trips (no pickle), shuffled/subset/new-level scoring, typed errors for
every dim policy, context-window equivalence, evaluate == score path, output units,
aggregation legality, seed determinism and the About card. Every check must pass before you
share the plugin.
"""

from __future__ import annotations

import pytest
from mapi_sdk.testing import ModelFixture, check_model, toy

from acme.model import MODELS, GroupMeanModel

SPEC = """
apiVersion: mapi/v1
kind: ModelSpec
metadata: {name: contract-kit, project: test}
data:
  dataset: dataset://test/panel@latest
  bind:
    dims: {geo: market, time: date}
    vars: {y: revenue_local}
  split: {kind: time_holdout, dim: time, last: 4, after_step: weekly}
pipeline:
  - {id: weekly, use: toy.resample@^1, with: {dim: time, freq: W-SUN}}
model:
  use: acme.group_mean@^0.1
  settings: {shrinkage: 0.1}
  fit: {mode: evaluate_then_refit, seed: 7}
evaluate: {metrics: [rmse, mape]}
"""


def test_the_model_is_exported() -> None:
    assert GroupMeanModel in MODELS
    assert GroupMeanModel.catalog_entry().ref == "acme.group_mean@0.1.0"


@pytest.mark.mapi_contract
def test_contract_kit() -> None:
    data, _ = toy.panel_dataset(days=7 * 24)
    fixture = ModelFixture(spec=SPEC, data=data[["revenue_local"]], registry=toy.toy_registry())
    report = check_model(GroupMeanModel, fixture, raise_on_failure=False, require_about=True)
    assert report.ok, report.summary()

    check_model fits the model on a small panel, then checks bundle round trips (no pickle), scoring in another order, on a subset and with new levels, typed errors for every dim policy, the evaluate path, output units, seed determinism and the About card. See Contract kits.

    Use it¶

    In a spec:

    model:
  use: acme.group_mean@^0.1
  settings: {shrinkage: 0.1}

    Once it is installed on your stack, the model is in Catalog › Yours with an "Installed by your team" badge, in the model picker of Build, in the Model Explorer and in dashboard presets. Analysts can also make a preset over it with no code.
